async function loginWithCredentials() { const region = document.getElementById('region').value; const clientId = document.getElementById('implicitClientId').value; // Usando el mismo campo const clientSecret = document.getElementById('clientSecret').value; if (!clientId || !clientSecret) { showError('Por favor ingresa Client ID y Client Secret'); return; } try { // Llamamos a tu backend para obtener el token const response = await fetch('http://localhost:3000/token', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ region, clientId, clientSecret }) }); const data = await response.json(); if (data.access_token) { // Guardamos todo igual que antes para no romper grabaciones.html ni app.html sessionStorage.setItem('gc_token', data.access_token); sessionStorage.setItem('gc_region', region); sessionStorage.setItem('gc_method', 'credentials'); window.location.href = 'app.html'; } else { showError('Error de autenticación: ' + (data.error || 'Verifica tus credenciales')); } } catch (err) { showError('No se pudo conectar con el servidor Node.js'); } }
