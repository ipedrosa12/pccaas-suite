async function loginWithCredentials() { const region = document.getElementById('region').value; const clientId = document.getElementById('clientId').value; const clientSecret = document.getElementById('clientSecret').value; if (!clientId || !clientSecret) { showError('Debes ingresar Client ID y Secret'); return; } const errorBox = document.getElementById('errorBox'); errorBox.style.display = 'none'; try { // Usamos 127.0.0.1 que es la dirección física local más fiable const response = await fetch('http://127.0.0.1:3000/token', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ region, clientId, clientSecret }) }); const data = await response.json(); if (data.access_token) { sessionStorage.setItem('gc_token', data.access_token); sessionStorage.setItem('gc_region', region); sessionStorage.setItem('gc_method', 'credentials'); window.location.href = 'app.html'; } else { showError('Error de Genesys: ' + (data.error || 'Credenciales no válidas')); } } catch (err) { console.error(err); showError('🔌 No hay conexión con el servidor local. 1. Ejecuta "node server.js". 2. Verifica que el puerto 3000 esté libre.'); } }
